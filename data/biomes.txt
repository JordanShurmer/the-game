# Biome table for the Noita-like game
# Format matches materials.txt, so one small parser reads both.
#
# Rules:
#   - Lines starting with # are comments
#   - Blank lines are ignored
#   - A biome starts with [Name]
#   - Fields are key = value
#   - [Map] is a reserved section that holds the world map settings
#   - [Laboratory] is the other one, and it holds a second world
#
# Every key color must be unique. The loader reports a duplicate as a
# hard error. Each key color must also appear on one of the two map
# pictures, where one pixel is one region of the world. No biome may be
# named Map or Laboratory, because those two names are already taken.

[Map]
image           = data/biome_map.png
origin_pixel    = 8 8         # map pixel that holds world cell (0,0)
cells_per_pixel = 512         # world cells along one edge of a region
seed            = 20260818    # lays out the tile lattice; change it for another world
biome_off_map   = Deep_Rock   # biome used outside the map image
spawn_biome     = Homelands   # the wizard starts in this biome
spawn_region    = 4           # the fourth of its regions, west to east

# The seed above is the one the game starts on. `seed=N` on the command
# line gives another, and every binary that opens the world takes it:
#
#   ./bin/the-game seed=7
#   ./bin/shot seed=7 biome=Coalmine out=shots/other.png
#   ./bin/bench seed=7 biome=Coalmine
#   ./bin/game-mcp seed=7
#
# A seed lays this map out: which tile of a wang set each square of the
# lattice draws, and which of an image biome's pictures each of its
# regions draws. The map itself does not change, so every seed is the
# same world drawn again.
#
# One seed is not that. See [Laboratory] below.

# [Laboratory] is the second reserved section, and it is a second
# world rather than a biome. One seed does not lay the map above out
# again: it opens another map picture altogether, with its own spawn.
# That world is the museum -- the two galleries and the roof between
# their doors, and nothing else. See docs/laboratory.md.
#
#   ./bin/the-game seed=0x1AB
#
# 0x1AB is LAB, in the one alphabet a number has.

[Laboratory]
seed         = 0x1AB
image        = data/biome_map_laboratory.png
spawn_biome  = Gallery        # he lands on its roof, between the two doors
spawn_region = 1

# cells_per_pixel must be a whole number of tiles, or a biome border
# would cut a tile in half. A tile is 512 cells, so that is a multiple
# of 512, and at 512 a region is exactly one tile.

# A biome may also be a light:
#
#   light = <material>    the biome throws that material's luminosity
#
# The sky throws Daylight. Everything under it is lit by the day
# reaching it, which is what makes a field bright, a wood dim and a
# cave dark, and it costs the world no light source at all. See
# docs/lighting.md, "The day is a biome".

# A biome fills its regions one of three ways.
#
#   generator = uniform   the whole region is fill_0
#   generator = wang      the region is a lattice of tiles from a set
#   generator = image     the region is one whole painted picture
#
# A tile is a 512x512 PNG painted in material colors, one cell to one
# world cell. A wang biome owns a whole set of them, named after the
# prefix below:
#
#   <tiles>_<N><E><S><W>_<variant>.png
#
# The four digits are the colors of the four edges of that tile, north
# east south west. The world only ever puts two tiles side by side
# when they agree about the edge they share, so the pattern runs on
# with no seam in it. There are two colors, so a complete set is 16
# tiles, times the variants: several drawings of one set of edges,
# picked at random, so the same four edges do not always give the same
# picture.
#
# The cells within 4 of a side belong to the edge color rather than
# to the tile, and the editor writes a stroke there into every tile that
# carries that color. Press T in the world editor to paint a set.
#
# generator = wang and tiles go together: either one on its own is a
# hard error. fill_0 stays required, because a tile with no file yet
# starts as a flat block of it.
#
# An image biome names a prefix the same way, and its pictures are
#
#   <image>_<variant>.png
#
# each one cells_per_pixel square. variants says how many there are,
# and the world picks one per region off the world seed, so a biome
# several regions long is several drawings of its set. One picture is
# the ordinary case and the file is <image>_0.png.

[Sky]
